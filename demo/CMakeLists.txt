# Generate the compiled-in demo registry (title|executable|description per
# demo, collected by uise_demo() in cmake/uisedemo.cmake) for demomanager.
# All ADD_SUBDIRECTORY() calls above must run first so the registry is complete.

GET_PROPERTY(uiseDemoRegistry GLOBAL PROPERTY UISE_DEMO_REGISTRY)
IF (uiseDemoRegistry)
    LIST(SORT uiseDemoRegistry)
ENDIF()

SET(demoRegistryContent "// Auto-generated by demo/CMakeLists.txt - do not edit\n")
FOREACH (entry ${uiseDemoRegistry})
    STRING(REPLACE "|" ";" entryFields "${entry}")
    LIST(GET entryFields 0 entryTitle)
    LIST(GET entryFields 1 entryExe)
    LIST(GET entryFields 2 entryDescription)

    # Escape for embedding into a C++ string literal (backslash first, then quote).
    STRING(REPLACE "\\" "\\\\" entryTitle "${entryTitle}")
    STRING(REPLACE "\"" "\\\"" entryTitle "${entryTitle}")
    STRING(REPLACE "\\" "\\\\" entryDescription "${entryDescription}")
    STRING(REPLACE "\"" "\\\"" entryDescription "${entryDescription}")

    STRING(APPEND demoRegistryContent "    { \"${entryExe}\", \"${entryTitle}\", \"${entryDescription}\" },\n")
ENDFOREACH()

SET(demoRegistryGenDir ${CMAKE_CURRENT_BINARY_DIR}/demomanager/generated)
FILE(MAKE_DIRECTORY ${demoRegistryGenDir})
FILE(WRITE ${demoRegistryGenDir}/demoregistry.inc.tmp "${demoRegistryContent}")
CONFIGURE_FILE(${demoRegistryGenDir}/demoregistry.inc.tmp ${demoRegistryGenDir}/demoregistry.inc COPYONLY)

ADD_SUBDIRECTORY(demomanager)
