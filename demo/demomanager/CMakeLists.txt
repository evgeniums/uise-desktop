CMAKE_MINIMUM_REQUIRED (VERSION 3.16)

# Not built via uise_demo(): this is the launcher itself, not a demo, and it
# owns the bundle/plist handling that uise_demo() applies to its siblings.
SET(demoManagerTarget ${UISE_DEMO_MANAGER_NAME})

SET (HEADERS
    ${CMAKE_CURRENT_SOURCE_DIR}/demomanagerwindow.hpp
    ${CMAKE_CURRENT_SOURCE_DIR}/demoregistry.hpp
)

SET (SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/main.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/demomanagerwindow.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/demoregistry.cpp
)

ADD_EXECUTABLE(${demoManagerTarget} ${HEADERS} ${SOURCES})
TARGET_LINK_LIBRARIES(${demoManagerTarget} PRIVATE ${UISE_DESKTOP_LIB_TARGET})

# demoregistry.inc is generated by demo/CMakeLists.txt into this exact directory.
TARGET_INCLUDE_DIRECTORIES(${demoManagerTarget} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

SET_TARGET_PROPERTIES(${demoManagerTarget} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${UISE_DEMO_BIN_DIR})
FOREACH (cfg Debug Release RelWithDebInfo MinSizeRel)
    STRING(TOUPPER ${cfg} cfgUpper)
    SET_TARGET_PROPERTIES(${demoManagerTarget} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${cfgUpper} ${UISE_DEMO_BIN_DIR})
ENDFOREACH()

IF (UISE_DESKTOP_DEMO_BUNDLE)

    IF (WIN32)
        SET_TARGET_PROPERTIES(${demoManagerTarget} PROPERTIES WIN32_EXECUTABLE TRUE)
    ENDIF()

    IF (APPLE)
        SET(MACOSX_BUNDLE_BUNDLE_NAME "UISE Demos")
        SET(MACOSX_BUNDLE_GUI_IDENTIFIER "org.uise.desktop.demomanager")
        SET(MACOSX_BUNDLE_EXECUTABLE_NAME ${demoManagerTarget})
        SET(MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION})
        SET(MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
        SET(MACOSX_BUNDLE_LONG_VERSION_STRING ${PROJECT_VERSION})
        SET(MACOSX_BUNDLE_INFO_STRING "UISE Demos")
        SET(MACOSX_BUNDLE_COPYRIGHT "")
        SET(MACOSX_BUNDLE_ICON_FILE "")

        SET_TARGET_PROPERTIES(${demoManagerTarget} PROPERTIES
            MACOSX_BUNDLE TRUE
            MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/macos/Info.plist.in
        )
    ENDIF()

ENDIF()
